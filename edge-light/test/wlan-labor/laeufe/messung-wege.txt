# Messung (ohne Urteil) zum Lauf eigene-wege: warum fehlt der Box der eigene Weg zu einem Geraet an ihrer LAN-Buchse, obwohl relayd es fuer die Weiterleitung fuehrt? Kernel (Absender, der der Box nicht gehoert) und Zeitleiste relayd/WLAN-Adresse bei Netz-Neustart und ifup
@180 printf 'kundenpasswort\n' | $W verbinden Kunde-WLAN > /tmp/v.out; echo "rc=$?"; WWANIP=$(wert wlan_adresse /tmp/v.out); echo "wlan=$WWANIP"
@60 KABELIP=$(dhcp_neu kabel kabel1); echo "kabelgeraet=$KABELIP"; ip netns exec kabel sh -c 'while :; do ping -c 1 -W 1 192.168.178.1 >/dev/null 2>&1; sleep 4; done' >/dev/null 2>&1 & echo $! > /tmp/geraet-redet.pid
# Kernel 1: Absender, der der Box (noch) nicht gehoert - in einer eigenen Tabelle wie in der Haupttabelle abgewiesen
info "Kernel, fremder Absender: eigene Tabelle: $(ip route add 192.168.178.252/32 dev br-lan src 192.168.178.254 table 16899 proto boot scope link 2>&1 || true) / Haupttabelle: $(ip route add 192.168.178.252/32 dev br-lan src 192.168.178.254 2>&1 || true)"; ip route flush table 16899
# Kernel 2: verschwindet die Absenderadresse, raeumt der Kernel nur die Haupttabelle - ein Weg in relayds Tabelle bleibt stehen
ip addr add 192.168.178.251/32 dev br-lan; ip route add 192.168.178.252/32 dev br-lan src 192.168.178.251 table 16899; ip route add 192.168.178.253/32 dev br-lan src 192.168.178.251; ip addr del 192.168.178.251/32 dev br-lan; info "Kernel, Absender verschwindet: eigene Tabelle [$(ip route show table 16899 | tr -s ' ')] Haupttabelle [$(ip route show | grep -F 192.168.178.253 | tr -s ' ')]"; ip route flush table 16899
# Zeitleiste je Sekunde (werkzeug/zeitleiste.sh): relayd (PIDs), WLAN-Adresse, vp_wlan oben, Geraet in der Weiterleitung und im eigenen Weg der Box
@180 sleep 20; sh /tmp/labor/werkzeug/zeitleiste.sh $KABELIP > /tmp/zeit1.txt 2>&1 & echo $! > /tmp/zeit.pid; sleep 2; /etc/init.d/network restart; sleep 45; kill $(cat /tmp/zeit.pid); awk '{k=$0; sub(/^[^ ]* /, "", k); if (k != v) print; v = k}' /tmp/zeit1.txt | sed 's/^/INFO Neustart /'
@180 sleep 10; sh /tmp/labor/werkzeug/zeitleiste.sh $KABELIP > /tmp/zeit2.txt 2>&1 & echo $! > /tmp/zeit.pid; sleep 2; ifup vp_wlan; sleep 45; kill $(cat /tmp/zeit.pid); awk '{k=$0; sub(/^[^ ]* /, "", k); if (k != v) print; v = k}' /tmp/zeit2.txt | sed 's/^/INFO ifup /'
info "danach: relayd fuehrt das Geraet fuer die Weiterleitung: $(ip route show table 16801 | grep -c "^$KABELIP "), eigener Weg der Box: $(ip route show table 16800 | grep -c "^$KABELIP "), wege_fehlen=$($W status | sed -n s/^wege_fehlen=//p)"
